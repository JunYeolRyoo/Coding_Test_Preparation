``` python
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def reverseBetween(self, head, left, right):
        if left == right: return head
        dummy = ListNode(-1,head)
        curind = 1
        prev,cur = dummy,head

        while curind <= right:
            if curind == left:
                curprev = None
                while curind <= right:
                    next = cur.next
                    cur.next = curprev
                    curprev = cur
                    cur = next
                    curind += 1
                prev.next.next = cur
                prev.next = curprev
                return dummy.next
            prev = cur
            cur = cur.next
            curind += 1
```

***Time Complexity: O(N)<br>***
***Space Complexity: O(1)***
